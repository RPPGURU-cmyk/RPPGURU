        function delay(ms) {
            return new Promise(resolve => setTimeout(resolve, ms));
        }

        // Membersihkan response JSON dari markdown block
        function cleanJsonText(text) {
            if (!text) return "{}";
            return text.replace(/```json\s*/g, "").replace(/```\s*$/g, "").trim();
        }

        async function fetchWithBackoff(url, options, retries = 3) {
            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (!response.ok) {
                        if (response.status === 429 && i < retries - 1) {
                            await delay(Math.pow(2, i) * 1000);
                            continue;
                        }
                        throw new Error(`API error: ${response.statusText}`);
                    }
                    return response.json();
                } catch (error) {
                    if (i === retries - 1) throw error;
                    await delay(1000);
                }
            }
        }

        function getLoadingHTML(text) {
            return `<div class="loading-text"><span class="spinner spinner-dark"></span>${text}</div>`;
        }

        function markdownToHtml(text) {
            if (!text) return '<p></p>';
            let html = text
                .replace(/^##\s*(.*?)$/gm, '<h5 class="font-bold text-lg mt-4 mb-2 text-indigo-800">$1</h5>')
                .replace(/^###\s*(.*?)$/gm, '<h6 class="font-bold text-base mt-3 mb-1 text-gray-800">$1</h6>')
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                .replace(/\*(.*?)\*/g, '<em>$1</em>');

            return html.split('\n\n').map(paragraph => {
                if (paragraph.trim().length === 0) return '';
                if (paragraph.trim().startsWith('<h')) return paragraph;
                
                const lines = paragraph.split('\n');
                if (lines.length > 1 && /^\s*[\d+\.\-\*]/.test(lines[0].trim())) {
                    const isOrdered = /^\s*\d+\./.test(lines[0].trim());
                    const listTag = isOrdered ? 'ol' : 'ul';
                    const listClass = isOrdered ? 'list-decimal' : 'list-disc';
                    const items = lines.map(line => `<li>${line.replace(/^\s*[\d+\.\-\*]\s*/, '')}</li>`).join('');
                    return `<${listTag} class="${listClass} list-inside space-y-1 pl-4 mb-3">${items}</${listTag}>`;
                }
                return `<p class="mb-3 text-justify">${paragraph.replace(/\n/g, '<br>')}</p>`;
            }).join('');
        }

        async function generateDesignDetails(mapel, jenjang, kelas, topik, tujuan) {
            const userQuery = `Analisis desain pembelajaran: Mapel "${mapel}", Kelas ${kelas} (${jenjang}), Topik "${topik}", TP "${tujuan}".
            Berikan analisis singkat (2-3 kalimat per poin) untuk:
            1. Peserta Didik
            2. Materi Pelajaran
            3. Lintas Disiplin
            4. Kemitraan
            5. Lingkungan
            6. Digital`;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: DESIGN_SCHEMA,
                    temperature: 0.4,
                }
            };

            try {
                const data = await fetchWithBackoff(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
                return JSON.parse(cleanJsonText(text));
            } catch (e) {
                console.error(e);
                return { peserta_didik: "Gagal memuat.", materi_pelajaran_analisis: "Gagal memuat.", lintas_disiplin: "-", kemitraan: "-", lingkungan: "-", digital: "-" };
            }
        }

        async function generateQuestions(topik, tujuan) {
            const userQuery = `Buat 15 soal asesmen sumatif (10 Pilihan Ganda dengan opsi ABCD, 5 Isian Singkat) beserta kunci jawaban untuk topik "${topik}" dan tujuan "${tujuan}". Output JSON murni.`;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: QUESTIONS_SCHEMA,
                    temperature: 0.5,
                }
            };

            try {
                const data = await fetchWithBackoff(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
                return JSON.parse(cleanJsonText(text));
            } catch (e) {
                console.error(e);
                return [];
            }
        }

        async function generateTextContent(prompt, temp = 0.5) {
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                generationConfig: { temperature: temp }
            };
            try {
                const data = await fetchWithBackoff(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                return data.candidates?.[0]?.content?.parts?.[0]?.text || "Gagal menghasilkan konten.";
            } catch (e) {
                console.error(e);
                return "Terjadi kesalahan saat menghubungi AI.";
            }
        }

        // Helper untuk Kegiatan Pembelajaran Statis (Agar struktur terjaga)
        function generateMeetingBreakdown(model, topik, tujuan) {
            let syntax = "";
            if (model === 'pbl') {
                syntax = `
                    <li><strong>Orientasi Masalah:</strong> Guru menyajikan masalah kontekstual terkait ${topik}.</li>
                    <li><strong>Organisasi Belajar:</strong> Siswa berkelompok mendefinisikan masalah.</li>
                    <li><strong>Penyelidikan:</strong> Siswa mencari data/referensi untuk solusi.</li>
                    <li><strong>Penyajian:</strong> Kelompok mempresentasikan hasil diskusi.</li>
                    <li><strong>Evaluasi:</strong> Guru dan siswa menyimpulkan pemecahan masalah.</li>
                `;
            } else if (model === 'pjbl') {
                syntax = `
                    <li><strong>Pertanyaan Mendasar:</strong> Guru memberikan pertanyaan tantangan tentang ${topik}.</li>
                    <li><strong>Desain Proyek:</strong> Siswa menyusun rencana dan jadwal proyek.</li>
                    <li><strong>Monitoring:</strong> Guru memantau progres pembuatan produk.</li>
                    <li><strong>Uji Hasil:</strong> Mempresentasikan produk/karya.</li>
                    <li><strong>Evaluasi Pengalaman:</strong> Refleksi proses pembuatan proyek.</li>
                `;
            } else {
                syntax = `
                    <li><strong>Stimulasi:</strong> Guru memberikan apersepsi menarik tentang ${topik}.</li>
                    <li><strong>Identifikasi:</strong> Siswa merumuskan pertanyaan.</li>
                    <li><strong>Pengumpulan Data:</strong> Eksplorasi materi dan diskusi.</li>
                    <li><strong>Verifikasi:</strong> Pembuktian konsep melalui latihan/presentasi.</li>
                    <li><strong>Generalisasi:</strong> Menarik kesimpulan bersama.</li>
                `;
            }

            return `
                <div class="space-y-4">
                    <div class="border-l-4 border-indigo-500 pl-4 py-2 bg-indigo-50">
                        <h5 class="font-bold text-indigo-800">Pendahuluan (15%)</h5>
                        <ul class="list-disc list-inside text-sm text-gray-700">
                            <li>Guru membuka dengan salam, doa, dan absensi.</li>
                            <li>Apersepsi: Mengaitkan materi dengan pengalaman siswa.</li>
                            <li>Menyampaikan Tujuan Pembelajaran: ${tujuan}.</li>
                            <li>Ice Breaking (Lihat Lampiran).</li>

                        <ol class="list-decimal list-inside text-sm text-gray-700 space-y-1">
                            ${syntax}
                        </ol>
                    </div>
                    <div class="border-l-4 border-green-500 pl-4 py-2 bg-green-50">

                            <li>Penyampaian agenda pertemuan berikutnya dan doa penutup.</li>
                        </ul>
                    </div>
                </div>
            `;
        }

        // FUNGSI DOWNLOAD WORD
        function downloadWord() {
            const content = document.getElementById('plan-output').innerHTML;
            const topik = document.getElementById('topik_pembelajaran').value || "RPP";
            
            const preHtml = `<html xmlns:o='urn:schemas-microsoft-com:office:office' xmlns:w='urn:schemas-microsoft-com:office:word' xmlns='http://www.w3.org/TR/REC-html40'><head><meta charset='utf-8'><title>Export HTML to Word Document with JavaScript</title>
            <style>
                body { font-family: 'Times New Roman', serif; font-size: 12pt; }
                h1, h2, h3, h4 { font-family: Arial, sans-serif; font-weight: bold; }
                h3 { border-bottom: 1px solid #000; padding-bottom: 5px; margin-top: 20px; }
                table { border-collapse: collapse; width: 100%; margin-bottom: 10px; }
                td { vertical-align: top; padding: 5px; }
                .bg-gray-50, .bg-blue-50, .bg-yellow-50, .bg-green-50 { background-color: transparent; border: 1px solid #ccc; padding: 10px; margin-bottom: 10px; }
                /* Menghapus elemen visual web saat di Word */
                .loading-text, .spinner { display: none; }
            </style></head><body>`;
            
            const postHtml = "</body></html>";
            const html = preHtml + content + postHtml;

            const blob = new Blob(['\ufeff', html], { type: 'application/msword' });
            const url = 'data:application/vnd.ms-word;charset=utf-8,' + encodeURIComponent(html);
            
            const downloadLink = document.createElement("a");
            document.body.appendChild(downloadLink);
            
            if(navigator.msSaveOrOpenBlob ){
                navigator.msSaveOrOpenBlob(blob, `RPP_${topik}.doc`);
            } else {
                downloadLink.href = url;
                downloadLink.download = `RPP_${topik}.doc`;
                downloadLink.click();
            }
            document.body.removeChild(downloadLink);
        }
